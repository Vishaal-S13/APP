R PRACTICALS – USER INPUT PROGRAMS
1. Input name & age + R version
name <- readline("Enter name: ")
age <- readline("Enter age: ")
cat("Name:", name, "Age:", age)
R.version.string

2. Objects in memory
a <- as.numeric(readline("Enter number: "))
b <- readline("Enter string: ")
ls()

3. Vectors to Matrix
a <- scan(n=3)
b <- scan(n=3)
c <- scan(n=3)
cbind(a,b,c)

4. Data Frame from user input
name <- scan(what="character")
age <- scan()
data.frame(name, age)

5. Array creation
values <- scan()
dims <- scan(n=2)
array(values, dims)

6. List with heterogeneous data
chars <- scan(what="character")
nums <- scan()
logs <- scan(what=logical)
list(chars, nums, logs)

7. String manipulation
s <- readline("Enter string: ")
paste(s, "R")
nchar(s)
toupper(s)
substr(s,1,5)

8. File handling
name <- scan(what="character")
marks <- scan()
df <- data.frame(name, marks)
write.csv(df,"data.csv")
read.csv("data.csv")

9. Pie and Bar chart
values <- scan()
labels <- scan(what="character")
pie(values, labels)
barplot(values, names.arg=labels)

10. Bar plot of subject marks
marks <- scan()
subjects <- scan(what="character")
barplot(marks, names.arg=subjects)
